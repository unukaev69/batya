    <script>
        let count = 0;
        let isSnoring = false;

        const quotes = [
            "«Я не сплю, я просто глаза прикрыл, телевизор слушаю!»",
            "«Выключи свет, мы что, за электричество мало платим?!»",
            "«В моё время такой ерунды не было...»",
            "«Майонеза много не бывает.»",
            "«Кто последний пульт брал? Положите на место!»",
            "«Я не толстый, это комок нервов и авторитета.»",
            "«Сейчас полежу пять минут и пойду дела делать...»"
        ];

        function feedBatya() {
            count++;
            document.getElementById('pelmeniCount').innerText = count;
            
            let scale = 1 + (count * 0.005);
            if (scale > 1.25) scale = 1.25;
            document.getElementById('photoFrame').style.transform = `scale(${scale})`;

            const rankEl = document.getElementById('rank');
            if (count >= 50) rankEl.innerText = "Абсолютный гигачад дивана 🏆";
            else if (count >= 30) rankEl.innerText = "Тяжёлая артиллерия 🚜";
            else if (count >= 15) rankEl.innerText = "Солидный мужчина 😎";
            else if (count >= 5) rankEl.innerText = "Разогревается 🥟";
        }

        function newQuote() {
            const randomIndex = Math.floor(Math.random() * quotes.length);
            document.getElementById('quoteBox').innerText = quotes[randomIndex];
        }

        function toggleSnore() {
            const card = document.getElementById('mainCard');
            const btn = document.getElementById('snoreBtn');
            isSnoring = !isSnoring;

            if (isSnoring) {
                card.classList.add('shaking');
                btn.innerText = "🛑 Разбудить батю (ХРРР-ПШШШ)";
            } else {
                card.classList.remove('shaking');
                btn.innerText = "💤 Включить режим храпа";
            }
        }
    </script>
